CREATE TABLE nguoi_dung (
    ma_nguoi_dung SERIAL PRIMARY KEY,
    ten_dang_nhap VARCHAR(100) UNIQUE NOT NULL,
    mat_khau VARCHAR(255) NOT NULL,
    ho_ten VARCHAR(150) NOT NULL,
    so_dien_thoai VARCHAR(20) UNIQUE,
    email VARCHAR(150) UNIQUE,
    trang_thai VARCHAR(30) DEFAULT 'HOAT_DONG'
);

CREATE TABLE khach_hang (
    ma_nguoi_dung INT PRIMARY KEY REFERENCES nguoi_dung(ma_nguoi_dung)
);

CREATE TABLE vai_tro (
    ma_vai_tro SERIAL PRIMARY KEY,
    ten_vai_tro VARCHAR(100) UNIQUE NOT NULL
);

CREATE TABLE nhan_vien (
    ma_nguoi_dung INT PRIMARY KEY REFERENCES nguoi_dung(ma_nguoi_dung),
    ma_vai_tro INT REFERENCES vai_tro(ma_vai_tro)
);

CREATE TABLE quyen (
    ma_quyen SERIAL PRIMARY KEY,
    ten_quyen VARCHAR(100) UNIQUE NOT NULL,
    mo_ta TEXT
);

CREATE TABLE vai_tro_quyen (
    ma_vai_tro INT REFERENCES vai_tro(ma_vai_tro),
    ma_quyen INT REFERENCES quyen(ma_quyen),
    PRIMARY KEY (ma_vai_tro, ma_quyen)
);

CREATE TABLE tinh (
    ten_tinh VARCHAR(100) PRIMARY KEY
);

CREATE TABLE chuyen_mon (
    ma_chuyen_mon SERIAL PRIMARY KEY,
    ten_chuyen_mon VARCHAR(100),
    mo_ta_chuyen_mon TEXT
);

CREATE TABLE loai_dich_vu (
    ma_loai_dich_vu SERIAL PRIMARY KEY,
    ten_loai_dich_vu VARCHAR(100),
    mo_ta_ldv TEXT
);

CREATE TABLE loai_khach_hang (
    ma_loai_khach_khach SERIAL PRIMARY KEY,
    ten_loai_khach_hang VARCHAR(100),
    muc_uu_tien INT
);

CREATE TABLE thoi_diem (
    ma_thoi_diem SERIAL PRIMARY KEY,
    ngay_bat_dau TIMESTAMP,
    ngay_ket_thuc TIMESTAMP
);

CREATE TABLE vat_tu (
    ma_vat_tu SERIAL PRIMARY KEY,
    ten_vat_tu VARCHAR(100),
    don_vi_tinh VARCHAR(50)
);

CREATE TABLE khu_vuc (
    ma_khu_vuc SERIAL PRIMARY KEY,
    ten_khu_vuc VARCHAR(100)
);

CREATE TABLE quan_huyen_tp (
    ma_quan_huyen SERIAL PRIMARY KEY,
    ten_thanh_pho VARCHAR(100),
    tp_tinh VARCHAR(100) REFERENCES tinh(ten_tinh)
);

CREATE TABLE phuong_xa (
    ma_phuong_xa SERIAL PRIMARY KEY,
    ten_phuong_xa VARCHAR(100),
    px_tp INT REFERENCES quan_huyen_tp(ma_quan_huyen)
);

CREATE TABLE duong (
    ma_duong SERIAL PRIMARY KEY,
    ten_duong VARCHAR(100),
    da_px INT REFERENCES phuong_xa(ma_phuong_xa)
);

CREATE TABLE dia_chi (
    ma_dia_chi SERIAL PRIMARY KEY,
    so_nha VARCHAR(50),
    dc_da INT REFERENCES duong(ma_duong)
);

CREATE TABLE kho_hang (
    ma_kho SERIAL PRIMARY KEY,
    loai_kho VARCHAR(50),
    ten_kho VARCHAR(100),
    ma_khu_vuc INT REFERENCES khu_vuc(ma_khu_vuc)
);

CREATE TABLE dich_vu (
    ma_dich_vu SERIAL PRIMARY KEY,
    ten_dich_vu VARCHAR(100),
    thoi_gian_xu_ly_du_kien INTERVAL,
    ma_loai_dich_vu INT REFERENCES loai_dich_vu(ma_loai_dich_vu)
);

CREATE TABLE don_gia_dich_vu (
    ma_dich_vu INT REFERENCES dich_vu(ma_dich_vu),
    ma_thoi_diem INT REFERENCES thoi_diem(ma_thoi_diem),
    don_gia_thoi_diem NUMERIC(15, 2),
    PRIMARY KEY (ma_dich_vu, ma_thoi_diem)
);

CREATE TABLE hop_dong (
    ma_hop_dong SERIAL PRIMARY KEY,
    ten_hop_dong VARCHAR(100),
    ngay_bat_dau_hop_dong DATE,
    ngay_ket_thuc_hop_dong DATE,
    chu_ky VARCHAR(50),
    ma_khach_hang INT REFERENCES khach_hang(ma_nguoi_dung)
);

CREATE TABLE phieu_bao_tri (
    ma_phieu_bao_tri SERIAL PRIMARY KEY,
    ten_phieu_bao_tri VARCHAR(100),
    ma_hop_dong INT REFERENCES hop_dong(ma_hop_dong)
);

CREATE TABLE phieu_yeu_cau (
    ma_phieu_yeu_cau SERIAL PRIMARY KEY,
    mo_ta_su_co TEXT,
    anh_su_co TEXT,
    trang_thai_phieu VARCHAR(50),
    thoi_gian_tao_phieu TIMESTAMP,
    khung_gio_mong_muon VARCHAR(100),
    ket_qua TEXT,
    ma_khach_hang INT REFERENCES khach_hang(ma_nguoi_dung),
    ma_dia_chi INT REFERENCES dia_chi(ma_dia_chi),
    ma_nhan_vien_tao INT REFERENCES nhan_vien(ma_nguoi_dung),
    ma_dich_vu INT REFERENCES dich_vu(ma_dich_vu)
);

CREATE TABLE hinh_anh_phieu (
    ma_anh SERIAL PRIMARY KEY,
    duong_dan TEXT,
    ma_phieu_yeu_cau INT REFERENCES phieu_yeu_cau(ma_phieu_yeu_cau)
);

CREATE TABLE hoa_don (
    ma_hoa_don SERIAL PRIMARY KEY,
    ngay_lap DATE,
    trang_thai_thanh_toan VARCHAR(50),
    hinh_thuc_thanh_toan VARCHAR(50),
    ma_hop_dong INT REFERENCES hop_dong(ma_hop_dong)
);

CREATE TABLE chi_tiet_hoa_don (
    ma_chi_tiet SERIAL PRIMARY KEY,
    don_gia NUMERIC(15, 2),
    thanh_tien NUMERIC(15, 2),
    ma_hoa_don INT REFERENCES hoa_don(ma_hoa_don),
    ma_phieu_yeu_cau INT REFERENCES phieu_yeu_cau(ma_phieu_yeu_cau)
);

CREATE TABLE vi_tri_hien_tai (
    ma_vi_tri SERIAL PRIMARY KEY,
    vi_tri VARCHAR(100),
    kinh_do NUMERIC(10, 6),
    thoi_gian_cap_nhat TIMESTAMP,
    ma_nhan_vien INT REFERENCES nhan_vien(ma_nguoi_dung)
);

CREATE TABLE phan_cong_ky_thuat (
    ma_phan_cong SERIAL PRIMARY KEY,
    thoi_gian_phan_cong TIMESTAMP,
    trang_thai VARCHAR(50),
    ghi_chu_cong_viec TEXT,
    ma_khu_vuc INT REFERENCES khu_vuc(ma_khu_vuc),
    ma_nhan_vien INT REFERENCES nhan_vien(ma_nguoi_dung)
);

CREATE TABLE khu_vuc_duoc_giao (
    ma_giao_khu_vuc SERIAL PRIMARY KEY,
    thoi_gian_giao TIMESTAMP,
    trang_thai VARCHAR(50),
    ly_do_tu_choi TEXT,
    ma_khu_vuc INT REFERENCES khu_vuc(ma_khu_vuc),
    ma_phieu_yeu_cau INT REFERENCES phieu_yeu_cau(ma_phieu_yeu_cau),
    ma_phan_cong INT REFERENCES phan_cong_ky_thuat(ma_phan_cong)
);

CREATE TABLE lich_lam_viec (
    ma_lich SERIAL PRIMARY KEY,
    ngay_lam DATE,
    ca_lam VARCHAR(50),
    ma_phan_cong INT REFERENCES phan_cong_ky_thuat(ma_phan_cong)
);

CREATE TABLE vat_tu_su_dung (
    ma_phan_cong INT REFERENCES phan_cong_ky_thuat(ma_phan_cong),
    ma_vat_tu INT REFERENCES vat_tu(ma_vat_tu),
    so_luong_su_dung INT,
    PRIMARY KEY (ma_phan_cong, ma_vat_tu)
);

CREATE TABLE ton_kho (
    ma_kho INT REFERENCES kho_hang(ma_kho),
    ma_vat_tu INT REFERENCES vat_tu(ma_vat_tu),
    so_luong_ton INT,
    PRIMARY KEY (ma_kho, ma_vat_tu)
);
